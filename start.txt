################## Making Exe as service #############################
Step 1: Check what kind of binary you have.
file ./ueba_client

Step 2: Create the installation directory
sudo mkdir -p /opt/ueba-client
sudo cp ./ueba_client /opt/ueba-client/
sudo chmod +x /opt/ueba-client/ueba_client

Step 3: Create the systemd service file

sudo tee /etc/systemd/system/ueba-client.service >/dev/null << 'EOF'
[Unit]
Description=UEBA Client Service
After=network.target

[Service]
ExecStart=/opt/ueba-client/ueba_client
Restart=always
RestartSec=3

[Install]
WantedBy=multi-user.target
EOF


to verify it: cat /etc/systemd/system/ueba-client.service

Step 4: Reload systemd + enable service at boot

sudo systemctl daemon-reload
sudo systemctl enable ueba-client
sudo systemctl start ueba-client

to verify: systemctl status ueba-client

TO stop the service:
sudo systemctl stop ueba-client  # To stop the service cleanly
sudo systemctl disable ueba-client # To disable it from starting on boot

or:
# 1) Stop old service (if running)
sudo systemctl stop ueba-client

# 2) Create /opt directory (if not already there)
sudo mkdir -p /opt/ueba-client

# 3) Copy new EXE to /opt and name it ueba_client
sudo cp /home/simar/Documents/UEBA/UEBA_Versions/UEBA_v1.2.2/ueba_client_1.2.2 /opt/ueba-client/ueba_client

# 4) Make it executable
sudo chmod 755 /opt/ueba-client/ueba_client

# 5) Copy updated service file
sudo cp /home/simar/Documents/UEBA/UEBA_Versions/UEBA_v1.2.2/ueba-client.service /etc/systemd/system/ueba-client.service
sudo chmod 644 /etc/systemd/system/ueba-client.service

# 6) Reload systemd and enable + start service
sudo systemctl daemon-reload
sudo systemctl enable ueba-client
sudo systemctl start ueba-client

# 7) Check status
systemctl status ueba-client
